USAGE EXAMPLES
═══════════════════════════════════════════════════════════════════════════════

Example 1: Basic Protection
────────────────────────────
router.post('/ai-chat/generate',
  authenticate,
  requireFeature('AI_CHAT_ENABLED'),
  controller.handler
);

Example 2: With Rate Limiting
──────────────────────────────
router.post('/quiz/generate',
  authenticate,
  rateLimitByTier({ freeLimit: 2, premiumLimit: 50 }),
  requireFeature('QUIZ_GENERATION_ENABLED'),
  controller.handler
);

Example 3: Multiple Features
─────────────────────────────
router.post('/analyze',
  authenticate,
  requireAnyFeature(['DOCUMENT_ANALYSIS_ENABLED', 'AI_CHAT_ENABLED']),
  controller.handler
);

Example 4: Conditional Display
───────────────────────────────
router.get('/dashboard',
  authenticate,
  attachFeatureInfo(),
  (req, res) => {
    const features = req.getAvailableFeatures();
    res.json({ success: true, features });
  }
);

Example 5: Admin Testing
────────────────────────
router.post('/admin/test',
  authenticate,
  adminFeatureOverride(),
  requireFeatureWithOverride('PREMIUM_FEATURE'),
  controller.handler
);
