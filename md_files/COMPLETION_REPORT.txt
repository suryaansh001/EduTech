   - Graceful error handling
   - Returns initialization status

   - Used for monitoring

5. shutdownAllServices()

USAGE EXAMPLE
================================================================================

// Controller implementation
import { createAIService, SUBSCRIPTION_TIERS } from '../services/ai/index.js';

export const chat = async (req, res) => {
  try {
    const userTier = req.user.subscriptionTier || SUBSCRIPTION_TIERS.FREE;
    const { message } = req.body;

    // Factory handles tier-based routing automatically
    const service = createAIService('chat', userTier, { logger });

    // Same interface regardless of tier
    const result = await service.sendMessage({
      userId: req.user.id,
      message
    });

    // Handle NoOp response for free tier
    if (!result.success) {
      return res.status(402).json({
        ...result,
        requiresUpgrade: userTier === SUBSCRIPTION_TIERS.FREE
      });
    }

    res.json({ success: true, data: result.data });
  } catch (error) {
    res.status(500).json({ success: false, message: error.message });
  }
};
